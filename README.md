    // 6. RENDER TOÀN BỘ 18 SẢN PHẨM VÀO 4 KHỐI DANH MỤC
    function renderAllProducts(filterType = 'all', searchKeyword = '') {
      const kw = searchKeyword.trim().toLowerCase();
      const filterBanner = document.getElementById('filterStatusBanner');
      const activeLabel = document.getElementById('activeFilterLabel');

      let filtered = productsData;

      if (kw !== '') {
        filtered = productsData.filter(p =>
          p.name.toLowerCase().includes(kw) ||
          p.category.toLowerCase().includes(kw) ||
          p.desc.toLowerCase().includes(kw)
        );
        if (filterBanner) filterBanner.classList.remove('hidden');
        if (activeLabel) activeLabel.textContent = `Từ khóa "${searchKeyword}" (${filtered.length} sản phẩm)`;
      } else if (filterType !== 'all') {
        if (filterType.startsWith('group-')) {
          filtered = productsData.filter(p => p.group === filterType);
        } else {
          filtered = productsData.filter(p => p.category.toLowerCase() === filterType.toLowerCase());
        }
        if (filterBanner) filterBanner.classList.remove('hidden');
        if (activeLabel) activeLabel.textContent = `${filterType} (${filtered.length} sản phẩm)`;
      } else {
        if (filterBanner) filterBanner.classList.add('hidden');
      }

      const groups = [
        { id: 'grid-group-sofa', secId: 'sec-sofa', groupKey: 'group-sofa' },
        { id: 'grid-group-bedroom', secId: 'sec-phongngu', groupKey: 'group-bedroom' },
        { id: 'grid-group-kitchen', secId: 'sec-banan', groupKey: 'group-kitchen' },
        { id: 'grid-group-living', secId: 'sec-ketivi', groupKey: 'group-living' }
      ];

      groups.forEach(g => {
        const gridEl = document.getElementById(g.id);
        const secEl = document.getElementById(g.secId);
        if (!gridEl || !secEl) return;

        const items = filtered.filter(p => p.group === g.groupKey);
        if (items.length === 0) {
          secEl.classList.add('hidden');
        } else {
          secEl.classList.remove('hidden');
          gridEl.innerHTML = items.map(item => createProductCardHTML(item)).join('');
        }
      });
    }

    // 7. SLIDER 2 BANNER TỰ TRƯỢT CHUYÊN NGHIỆP
    let currentSlideIndex = 0;
    let slideInterval = null;

    function showSlide(index) {
      const slides = document.querySelectorAll('.banner-slide');
      const dot0 = document.getElementById('sliderDot0');
      const dot1 = document.getElementById('sliderDot1');
      if (!slides.length) return;

      currentSlideIndex = (index + slides.length) % slides.length;
      slides.forEach((slide, i) => {
        if (i === currentSlideIndex) {
          slide.classList.remove('opacity-0', 'z-0');
          slide.classList.add('opacity-100', 'z-10');
        } else {
          slide.classList.remove('opacity-100', 'z-10');
          slide.classList.add('opacity-0', 'z-0');
        }
      });

      if (dot0 && dot1) {
        if (currentSlideIndex === 0) {
          dot0.className = 'w-5 h-1.5 rounded-full bg-amber-400 transition-all';
          dot1.className = 'w-2 h-1.5 rounded-full bg-white/60 transition-all';
        } else {
          dot0.className = 'w-2 h-1.5 rounded-full bg-white/60 transition-all';
          dot1.className = 'w-5 h-1.5 rounded-full bg-amber-400 transition-all';
        }
      }
    }

    function changeSlide(step) {
      showSlide(currentSlideIndex + step);
      resetSlideTimer();
    }

    function goToSlide(index) {
      showSlide(index);
      resetSlideTimer();
    }

    function resetSlideTimer() {
      if (slideInterval) clearInterval(slideInterval);
      slideInterval = setInterval(() => {
        showSlide(currentSlideIndex + 1);
      }, 5000);
    }

    // 9. GIỎ HÀNG & ĐẶT HÀNG NHANH
    let cart = [];

    function addToCart(productId) {
      const product = productsData.find(p => p.id === productId);
      if (!product) return;

      const existing = cart.find(item => item.id === productId);
      if (existing) {
        existing.qty += 1;
      } else {
        cart.push({ ...product, qty: 1 });
      }
      updateCartUI();
      showToast(`Đã thêm "${product.name}" vào giỏ hàng!`);
    }

    function removeFromCart(productId) {
      cart = cart.filter(item => item.id !== productId);
      updateCartUI();
    }

    function updateCartUI() {
      const totalCount = cart.reduce((sum, item) => sum + item.qty, 0);
      const badge = document.getElementById('cartCountBadge');
      if (badge) badge.textContent = totalCount;

      const container = document.getElementById('cartItemsContainer');
      const totalEl = document.getElementById('cartTotalPrice');
      if (!container || !totalEl) return;

      if (cart.length === 0) {
        container.innerHTML = `<p class="text-xs text-gray-500 text-center py-4">Giỏ hàng của quý khách đang trống. Hãy chọn sản phẩm cần mua!</p>`;
        totalEl.textContent = '0 ₫';
        return;
      }

      let totalAmount = 0;
      container.innerHTML = cart.map(item => {
        totalAmount += item.price * item.qty;
        return `
          <div class="flex items-center justify-between gap-2.5 p-2 rounded-lg bg-gray-50 border border-gray-200">
            <img src="${driveImg(item.driveId, 150)}" onerror="this.onerror=null;this.src='${driveFallback(item.driveId, 150)}';" alt="${item.name}" class="w-12 h-12 rounded object-cover shrink-0">
            <div class="flex-1 min-w-0">
              <div class="text-xs font-bold text-gray-800 line-clamp-1">${item.name}</div>
              <div class="text-[11px] text-brand-price font-bold">${formatVND(item.price)} x ${item.qty}</div>
            </div>
            <button onclick="removeFromCart(${item.id})" class="text-gray-400 hover:text-red-600 p-1">
              <i class="fa-solid fa-trash-can text-xs"></i>
            </button>
          </div>
        `;
      }).join('');

      totalEl.textContent = formatVND(totalAmount);
    }

    function openCartModal() {
      updateCartUI();
      document.getElementById('cartModal').classList.remove('hidden');
    }

    function closeCartModal() {
      document.getElementById('cartModal').classList.add('hidden');
    }

    function submitCartOrder(e) {
      e.preventDefault();
      if (cart.length === 0) {
        showToast('Vui lòng chọn ít nhất 1 sản phẩm trước khi đặt hàng!');
        return;
      }
      const name = document.getElementById('orderCustomerName').value;
      const phone = document.getElementById('orderCustomerPhone').value;
      cart = [];
      updateCartUI();
      closeCartModal();
      showToast(`Cảm ơn ${name} (${phone})! Nội Thất Salon Thắng sẽ gọi xác nhận giao hàng ngay.`);
    }
